# Rstest core

Core configuration and cleanup functions exported from `@rstest/core`.

## defineConfig

Provides type inference and excess-property checking for an Rstest configuration object or factory function.

- **Type:**

```ts
function defineConfig<
  const Config extends RstestConfig,
  const Definition extends Config | (() => Config) | (() => Promise<Config>),
>(
  config: Definition &
    (Definition extends (...args: never[]) => infer CallbackResult
      ? [Awaited<CallbackResult>] extends [RstestConfig]
        ? unknown
        : never
      : RstestConfig &
          Record<Exclude<keyof Definition, keyof RstestConfig>, never>),
): Definition extends (...args: never[]) => infer CallbackResult
  ? [CallbackResult] extends [RstestConfig]
    ? RstestConfigSyncFn
    : RstestConfigAsyncFn
  : RstestConfig;
function defineConfig<const Config extends RstestConfig>(
  config: () => Config,
): RstestConfigSyncFn;
function defineConfig<const Config extends RstestConfig>(
  config: () => Promise<Config>,
): RstestConfigAsyncFn;
function defineConfig(config: RstestConfig): RstestConfig;
function defineConfig(config: RstestConfigExport): RstestConfigExport;
```

- **Examples:**

Pass a config object directly:

```ts
import { defineConfig } from '@rstest/core';

export default defineConfig({
  testTimeout: 10000,
  coverage: {
    enabled: true,
    provider: 'istanbul',
  },
});
```

Return a config from a synchronous factory:

```ts
import { defineConfig } from '@rstest/core';

export default defineConfig(() => ({
  testTimeout: 10000,
}));
```

Return a config from an asynchronous factory:

```ts
import { defineConfig } from '@rstest/core';

export default defineConfig(async () => ({
  testTimeout: 10000,
}));
```

## defineProject

Provides type inference for an [Rstest Project](/guide/basic/projects) config file. Accepts a project config object, a nested `projects` object, or a sync/async factory function returning either.

- **Type:**

```ts
type NestedProjectConfig = {
  projects: (InlineProjectConfig | string)[];
};

function defineProject<
  const Config extends ProjectConfig | NestedProjectConfig,
>(
  config: () => Config,
): Config extends NestedProjectConfig
  ? () => NestedProjectConfig
  : () => ProjectConfig;
function defineProject<
  const Config extends ProjectConfig | NestedProjectConfig,
>(
  config: () => Promise<Config>,
): Config extends NestedProjectConfig
  ? () => Promise<NestedProjectConfig>
  : () => Promise<ProjectConfig>;
function defineProject(config: ProjectConfig): ProjectConfig;
function defineProject(config: NestedProjectConfig): NestedProjectConfig;
```

- **Examples:**

Pass a project config object directly:

```ts
import { defineProject } from '@rstest/core';

export default defineProject({
  root: './apps/web',
  testTimeout: 15000,
});
```

Return a project config from a synchronous factory:

```ts
import { defineProject } from '@rstest/core';

export default defineProject(() => ({
  root: './apps/web',
  testEnvironment: 'node',
}));
```

Return a project config from an asynchronous factory:

```ts
import { defineProject } from '@rstest/core';

export default defineProject(async () => ({
  root: './apps/web',
  testEnvironment: 'node',
}));
```

Define nested projects:

```ts
import { defineProject } from '@rstest/core';

export default defineProject({
  projects: [
    {
      name: 'web',
      root: './apps/web',
    },
    {
      name: 'server',
      root: './apps/server',
    },
  ],
});
```

Use `defineProject` for the top-level export of a project config file. Object items inside a returned `projects` array remain inline projects.

Unlike `defineInlineProject`, `defineProject` does not require `name` on the top-level project. When omitted, Rstest resolves `name` using the [name](/config/test/name) option rules: first from the project's `package.json`, then from its directory name.

## defineInlineProject

<ApiMeta addedVersion="0.9.6" />

Provides type inference for an inline project entry inside `defineConfig({ projects: [...] })`. Inline projects must specify `name` because they do not have a standalone config file to infer it from.

- **Type:**

```ts
function defineInlineProject(config: InlineProjectConfig): InlineProjectConfig;
```

- **Example:**

```ts
import { defineConfig, defineInlineProject } from '@rstest/core';

export default defineConfig({
  projects: [
    defineInlineProject({
      name: 'web',
      root: './apps/web',
      testTimeout: 15000,
    }),
  ],
});
```

## mergeRstestConfig

Deeply merges multiple Rstest config objects into a single config. Multiple function values are combined into sequentially executed functions (except under `browser`, which uses plain deep merge so arrays and functions are replaced). Later configs replace rather than concatenate `include`, `forceRerunTriggers`, `reporters`, and `coverage.reporters`.

- **Type:**

```ts
function mergeRstestConfig(...configs: RstestConfig[]): RstestConfig;
```

- **Example:**

```ts
import { mergeRstestConfig } from '@rstest/core';

const config = mergeRstestConfig(
  {
    testTimeout: 5000,
  },
  {
    coverage: {
      enabled: true,
    },
  },
);
```

## mergeProjectConfig

Deeply merges multiple project config objects into a single config using the same merge rules as `mergeRstestConfig`.

- **Type:**

```ts
function mergeProjectConfig(...configs: ProjectConfig[]): ProjectConfig;
```

- **Example:**

```ts
import { mergeProjectConfig } from '@rstest/core';

const config = mergeProjectConfig(
  {
    name: 'node',
    include: ['src/**/*.test.ts'],
  },
  {
    testTimeout: 10000,
  },
);
```

## loadConfig

Loads an Rstest config file and resolves any `extends` entries.

- **Type:**

```ts
function loadConfig(params?: {
  // Defaults to process.cwd()
  cwd?: string;
  // Config file path (relative or absolute)
  path?: string;
  envMode?: string;
  configLoader?: 'auto' | 'jiti' | 'native';
}): Promise<{
  content: RstestConfig;
  filePath: string | null;
  dependencies: string[];
}>;
```

- **Example:**

```ts
import { loadConfig } from '@rstest/core';

const { content, filePath } = await loadConfig({
  cwd: './my-project',
  path: './rstest.config.ts',
});

console.log('Config loaded from:', filePath);
console.log('Config:', content);
```

If no Rstest config file exists in `cwd`, `loadConfig` returns `{ content: {}, filePath: null, dependencies: [] }`.

## registerFileCleanup

<ApiMeta addedVersion="0.12.0" />

Registers a cleanup callback that runs when the current test file finishes in Node pools, including when test collection or suite execution fails. Returns a function that removes the callback before it runs.

Useful for Node-side test integrations that allocate per-file resources outside suite hooks. The registry is backed by a `Symbol.for` key on `globalThis`, so independently bundled runtime entries observe the same callbacks.

- **Type:**

```ts
function registerFileCleanup(cleanup: () => MaybePromise<void>): () => boolean;
```

- **Example:**

```ts
import { registerFileCleanup } from '@rstest/core';

const unregister = registerFileCleanup(async () => {
  await closePerFileResource();
});
```

## registerWorkerCleanup

<ApiMeta addedVersion="0.11.10" />

Registers a cleanup callback that runs when the current Node worker exits. Returns a function that removes the callback before it runs.

Useful for Node-side integrations that keep resources (such as browser instances or connections) alive across files when `isolate` is `false`. Like `registerFileCleanup`, the registry is backed by a `Symbol.for` key on `globalThis`.

- **Type:**

```ts
function registerWorkerCleanup(
  cleanup: () => MaybePromise<void>,
): () => boolean;
```

- **Example:**

```ts
import { registerWorkerCleanup } from '@rstest/core';

const unregister = registerWorkerCleanup(async () => {
  await closeSharedWorkerResource();
});
```
