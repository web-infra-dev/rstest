# exclude

- **Type:** `string[] | { patterns: string[]; override?: boolean }`
- **Default:** `['**/node_modules/**', '**/dist/**', '**/.{idea,git,cache,output,temp}/**']`
- **CLI:** `--exclude "**/fixtures/**"`

Glob patterns excluded from test file discovery, resolved relative to [root](/config/test/root). Rstest also automatically excludes its temporary build output directory ([`output.distPath.root`](/config/build/output#outputdistpath), `dist/.rstest-temp` by default).

## Example

Exclude test files under `fixtures`:

import { Tab, Tabs } from '@theme';

<Tabs defaultValue='rstest.config.ts'>
  <Tab label="CLI">
  ```bash
  npx rstest --exclude "**/fixtures/**"
  ```
  </Tab>
  <Tab label="rstest.config.ts">
```ts
import { defineConfig } from '@rstest/core';

export default defineConfig({
  exclude: ['**/fixtures/**'],
});
```
  </Tab>
</Tabs>

## Override default value

By default, Rstest merges custom `exclude` patterns with the defaults. Set `override: true` to replace the default patterns instead:

```ts title='rstest.config.ts'
import { defineConfig } from '@rstest/core';

export default defineConfig({
  exclude: {
    patterns: ['**/node_modules/**'],
    override: true,
  },
});
```
